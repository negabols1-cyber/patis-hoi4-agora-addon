##########################
#   Extra Spain Events   #
##########################

add_namespace = spain_extra

#Technological Advancement
country_event = {
	id = spain_extra.1
	title = spain_extra.1.t
	desc = spain_extra.1.desc
	picture = GFX_report_event_generic_factory

	fire_only_once = yes

	trigger = {
		original_tag = SPR
		OR = {
			AND = {
				OR = {
					has_completed_focus = SPA_the_condor_legion
					has_completed_focus = SPA_the_corpo_truppe_volontarie
				}
				OR = {
					has_government = neutrality
					has_government = fascism
				}
			}
			AND = {
				has_completed_focus = SPR_soviet_technological_advancements
				OR = {
					has_government = democratic
					has_government = communism
				}
			}
		}
		has_global_flag = scw_over
	}

	mean_time_to_happen = { days = 2 }

	option = {
		name = spain_extra.1.a
		if = { # German Tech
			limit = {
				has_completed_focus = SPA_the_condor_legion
			}
			if = {
				limit = { has_dlc = "No Step Back" }
				set_technology = {
					gwtank_chassis = 1
					basic_light_tank_chassis = 1
					basic_medium_tank_chassis = 1
					engine_tech_1 = 1
					armor_tech_1 = 1
				}
			}
			else = {
				set_technology = {
					gwtank = 1
					basic_light_tank = 1
					basic_light_td = 1
					basic_heavy_tank = 1
				}
			}
		}
		if = { # Italian Tech
			limit = {
				has_completed_focus = SPA_the_corpo_truppe_volontarie
			}
			if = {
				limit = { has_dlc = "By Blood Alone" }
				set_technology = {
					aa_lmg = 1
					engines_1 = 1
					engines_2 = 1
					iw_small_airframe = 1
					basic_small_airframe = 1
				}
			}
			else = {
				set_technology = {
					early_fighter = 1
					CAS1 = 1
					fighter1 = 1
					naval_bomber1 = 1
				}
			}
		}
		if = {
			limit = {
				has_completed_focus = SPR_soviet_technological_advancements
			}
			if = {
				limit = { has_dlc = "No Step Back" }
				set_technology = {
					gwtank_chassis = 1
					basic_heavy_tank_chassis = 1
					engine_tech_1 = 1
					armor_tech_1 = 1
				}
			}
			else = {
				set_technology = {
					gwtank = 1
					basic_heavy_tank = 1
					basic_heavy_td = 1
					basic_heavy_art = 1
				}
			}
		}
	}
}
