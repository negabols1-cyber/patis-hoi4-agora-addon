country_event = {
	id = liberia.10
	title = liberia.10.t
	desc = liberia.10.d

	picture = GFX_report_event_LIB_pearl_mining

	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			set_country_flag = {
				flag = LIB_waiting_for_answer_timeout
				days = 50
				value = 1
			}
		}
	}

	timeout_days = 30

	option = { #Sounds like a good idea
		name = liberia.10.a
		ai_chance = {
			base = 10
			modifier = {
				factor = 5
				OR = {
					original_tag = GER
					original_tag = FRA
					original_tag = USA
					original_tag = ENG
				}
				is_historical_focus_on = yes
			}
		}
		hidden_effect = {
			LIB = {
				country_event = {
					id = liberia.18
					hours = 3
				}
			}
		}
		custom_effect_tooltip = LIB_pearl_farming_event_tt
		hidden_effect = {
			random_state = {
				limit = {
					is_fully_controlled_by = ROOT
					industrial_complex > 0
				}
				remove_building = {
					type = industrial_complex
					level = 1
				}
				damage_building = {
					type = infrastructure
					damage = 1
				}
			}
		}
		effect_tooltip = {
			LIB_increase_pearl_trade = yes
		}
	}

	option = { #Can't trust those filthy neutrals
		name = liberia.10.b
		ai_chance = {
			base = 10
		}

		hidden_effect = {
			set_country_flag = {
				flag = LIB_rejected_investment
				days = 240
				value = 0
			}
		}
		LIB = {
			country_event = {
				id = liberia.19
				hours = 3
			}
		}
	}
}
