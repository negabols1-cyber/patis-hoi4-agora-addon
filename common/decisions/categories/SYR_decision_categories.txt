###########################################################################
# Syria (SYR) decision categories
#
# SYR_the_national_struggle: independence route (National Resistance meter,
#   the hour of revolt, the Damascus mission, leadership fallback).
# SYR_the_french_levant: loyalist route support for France at war.
# SYR_arabia_organization: Hashemite route, growth of the Arab-Islamic
#   alliance (after SYR_the_free_arab_nations) and union with Syria (after
#   SYR_the_arabia_organization).
# SYR_the_black_banner: secret route after SYR_komkompuration, the Day of
#   Reckoning countdown and the preparations for the war of faith; during the
#   war it keeps the aid decisions for the Black Banner side only. Its
#   description switches through the scripted localisation
#   SYR_GetBlackBannerStatus (common/scripted_localisation/SYR_scripted_localisation.txt).
# SYR_the_emirates_call: after SYR_the_call_to_the_arab_nations, demands that
#   the Arab nations submit as subjects of the Emirate (syria.106-108).
# Decisions: common/decisions/SYR.txt
# Design notes: docs/systems/syria_focus_tree.md
###########################################################################

SYR_the_national_struggle = {
	icon = GFX_decision_category_generic_independence
	priority = 100
	visible_when_empty = no
	allowed = {
		original_tag = SYR
	}
	visible = {
		OR = {
			has_completed_focus = SYR_crossroads_of_the_levant
			has_country_flag = SYR_treaty_betrayed
		}
	}
}

SYR_the_french_levant = {
	icon = GFX_decision_category_generic_arms_trade
	priority = 90
	visible_when_empty = no
	allowed = {
		original_tag = SYR
	}
	visible = {
		SYR_is_loyalist = yes
		SYR_is_french_mandate = yes
	}
}

SYR_arabia_organization = {
	icon = GFX_decision_category_gre_faction_management
	priority = 95
	visible_when_empty = yes
	allowed = {
		original_tag = SYR
	}
	visible = {
		OR = {
			has_completed_focus = SYR_the_free_arab_nations
			has_completed_focus = SYR_the_arabia_organization
		}
	}
}

SYR_the_emirates_call = {
	icon = GFX_decision_category_military_operation
	priority = 105
	visible_when_empty = no
	allowed = {
		original_tag = SYR
	}
	visible = {
		has_completed_focus = SYR_the_call_to_the_arab_nations
	}
}

SYR_the_black_banner = {
	icon = GFX_decision_category_generic_crisis
	priority = 110
	visible_when_empty = no
	allowed = {
		original_tag = SYR
	}
	visible = {
		has_country_flag = SYR_reckoning_begun
		NOT = { has_country_flag = SYR_war_of_faith_won }
		OR = {
			NOT = { has_country_flag = SYR_war_of_faith_started }
			has_country_flag = SYR_black_banner_side
		}
	}
}
