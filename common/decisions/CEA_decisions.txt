CEA_managing_the_banana_republic = {
	CEA_brewing_strikes = {
		priority = 15
		icon = revolt

		activation = {
			has_idea = CEA_the_banana_republic
			has_stability < 0.5
			NOT = {
				has_country_flag = CEA_improving_work_conditions_flag
			}
		}

		available = {
			OR = {
				has_country_flag = CEA_improving_work_conditions_flag
				has_stability > 0.49
			}
		}

		cancel_trigger = {
			NOT = {
				has_idea = CEA_the_banana_republic
			}
		}

		days_mission_timeout = 225

		timeout_effect = {
			add_timed_idea = {
				idea = strikes_1
				days = 60
			}

			if = {
				limit = {
					NOT = {
						has_country_flag = CEA_recent_strikes_flag
					}
				}
				set_country_flag = {
					flag = CEA_recent_strikes_flag
					days = 365
					value = 1
				}
			}

			custom_effect_tooltip = CEA_going_on_strikes_too_often_deter_USA_tt

			if = {
				limit = {
					has_country_flag = CEA_recent_strikes_flag
				}
				set_country_flag = {
					flag = CEA_USA_deterred_by_our_fragile_stability_flag
					days = 240
					value = 1
				}
			}
		}

		complete_effect = {
			add_stability = 0.01
		}
	}

	CEA_pay_off_the_debt = {
		priority = 20
		icon = eng_trade_unions_demand

		activation = {
			always = no
		}

		available = {
			NOT = {
				has_idea = CEA_repaying_the_american_debt
			}
		}

		cancel_trigger = {
			OR = {
				USA = {
					exists = no
					has_war_with = ROOT
				}
				NOT = {
					has_idea = CEA_repaying_the_american_debt
				}
			}
		}

		days_mission_timeout = 720

		timeout_effect = { ##might expand on this later on, depends on how lazy I am lol
			add_stability = -0.15
		}
	}

	CEA_redirect_fundings_to_improve_worker_conditions = {
		icon = generic_fundraising

		visible = {
			has_active_mission = CEA_brewing_strikes
			NOT = {
				has_country_flag = CEA_improving_work_conditions_flag
			}
		}

		cost = 20

		days_remove = 45

		modifier = {
			consumer_goods_factor = 0.2
			political_power_gain = -0.3
		}

		cancel_trigger = {
			NOT = {
				has_active_mission = CEA_brewing_strikes
			}
		}

		cancel_effect = {
			add_political_power = 20
		}

		remove_effect = {
			custom_effect_tooltip = CEA_worker_conditions_improved_for_year_tt
			hidden_effect = {
				set_country_flag = {
					flag = CEA_improving_work_conditions_flag
					days = 365
					value = 1
				}
			}
		}
	}

	CEA_request_additional_financing_from_the_USA = {
		icon = generic_welfare

		available = {
			NOT = {
				has_country_flag = CEA_USA_deterred_by_our_fragile_stability_flag
				has_idea = strikes_1
			}
			NOT = {
				has_country_flag = CEA_cant_ask_for_financing_flag
			}
		}

		visible = {
			has_idea = CEA_the_banana_republic
		}

		cost = CEA_request_financing_cost

		days_remove = 180

		days_re_enable = 180

		cancel_trigger = {
			OR = {
				NOT = {
					has_idea = CEA_the_banana_republic
				}
				USA = { exists = no }
				USA = { has_war_with = ROOT }
			}
		}

		modifier = {
			political_power_factor = 0.05
			consumer_goods_factor = -0.1
			production_speed_buildings_factor = 0.1
			industrial_capacity_factory = 0.1
		}

		remove_effect = {
			custom_effect_tooltip = EUYL_decision_will_reappear_in_180_days_tt

			if = {
				limit = {
					has_completed_focus = CEA_USA_root_economic_cooperation
				}
				EUYL_add_1_civilian = yes
			}
		}
	}

	CEA_improve_the_plantation_in_FROM = {
		icon = GFX_EUYL_banana_icon

		on_map_mode = map_and_decisions_view

		cost = 20

		state_target = yes

		target_array = controlled_states

		target_trigger = {
			FROM = {
				is_fully_controlled_by = ROOT
			}
		}

		visible = {
			has_completed_focus = CEA_improve_the_plantations
			has_idea = CEA_the_banana_republic
			FROM = {
				NOT = {
					has_dynamic_modifier = {
						modifier = CEA_improved_banana_production
					}
				}
			}
		}

		fire_only_once = yes

		available = {
			has_idea = CEA_the_banana_republic
			NOT = {
				has_idea = strikes_1
				has_country_flag = CEA_recently_bolstered_banana_production_flag
			}
		}

		complete_effect = {
			FROM = {
				if = {
					limit = {
						has_dynamic_modifier = {
							modifier = CEA_local_banana_production
						}
					}
					remove_dynamic_modifier = {
						modifier = CEA_local_banana_production
					}
					add_dynamic_modifier = {
						modifier = CEA_improved_banana_production
					}
				}
				else = {
					add_dynamic_modifier = {
						modifier = CEA_local_banana_production
					}
				}
			}
			set_country_flag = {
				flag = CEA_recently_bolstered_banana_production_flag
				days = 90
				value = 1
			}
		}
	}
