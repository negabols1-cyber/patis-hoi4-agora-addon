scripted_diplomatic_actions = {
	trade_deal= {
		allowed = { ROOT = { is_ai = no } }
		visible = {
			THIS = {NOT= {has_war_with = ROOT}}
			if = {
				limit = {
					has_game_rule = {
						rule = remove_all_multiplayer_tools
						option = ALLOWED
					}
				}
				always = no
			}
			NOT = {
				is_in_array = {
					array = ROOT.trade_deal
					value = FROM.id
				}
			}
		}

		selectable = {
			NOT = {
				is_in_array = {
					array = ROOT.embargoed_countries
					value = FROM.id
				}
			}
			if = {
				limit = {
					has_global_flag = EF_multiplayer_tools_available_flag
				}
				THIS = {is_ai = no}
			}
			NOT = { has_idea = closed_economy }
		}

		requires_acceptance = yes
		cost = 100
		command_power = 0

		show_acceptance_on_action_button = yes

		icon = 4

		on_sent_effect = {
		}

		can_be_accepted = {
			always = yes
		}
		can_be_sent = {
			always = yes
		}

		complete_effect = {
			td_diplomacy = yes

		}
		ai_acceptance = {
			condition = {
				base =10
			}
		}

		ai_desire = {
			base = 1

		}
	}

	trade_deal_reject= {
		allowed = { ROOT = { is_ai = no } }
		visible = {
			THIS = {NOT= {has_war_with = ROOT}}
			if = {
				limit = {
					has_game_rule = {
						rule = remove_all_multiplayer_tools
						option = ALLOWED
					}
				}
				always = no
			}
			THIS = { NOT = { has_opinion_modifier = cw_embargo_trade } }
			is_in_array = {
				array = ROOT.trade_deal
				value = FROM.id
			}
		}

		selectable = {

		}

		requires_acceptance = no
		cost = 0
		command_power = 0

		show_acceptance_on_action_button = no

		icon = 4

		on_sent_effect = {
		}

		can_be_accepted = {
			always = yes
		}
		can_be_sent = {
			always = yes
		}

		complete_effect = {
			td_diplomacy_remove = yes
		}
		ai_acceptance = {
			condition = {
				base =10
			}
		}

		ai_desire = {
			base = 1

		}
	}
}
