scripted_diplomatic_actions = {
	white_peace= {
		allowed = { ROOT = { is_ai = no } }
		visible = {
			THIS = {has_war_with = ROOT}
			if = {
				limit = {
					has_game_rule = {
						rule = remove_all_multiplayer_tools
						option = ALLOWED
					}
					NOT = { is_debug = yes }
				}
				always = no
			}
		}

		selectable = {
			if = {
				limit = {
					has_global_flag = EF_multiplayer_tools_available_flag
					NOT = { is_debug = yes }
				}
				THIS = {is_ai = no}
			}
			ROOT = {
				custom_trigger_tooltip = {
					tooltip = white_peace_flag_tt
				}
				check_variable = { ROOT.peace_var < 1 }
				has_political_power > 99
			}
		}

		requires_acceptance = yes
		cost = 100
		command_power = 0

		show_acceptance_on_action_button = yes

		icon = 4

		send_description = white_peace_desc

		reject_effect = {
			ROOT = {
				set_variable = {
					var = ROOT.peace_var
					value = 1
				}
			}
		}

		#reject_effect = {

					#		value = 1
					#	}
				#}
			#}

		complete_effect = {
			FROM = {
				white_peace = ROOT
			}
		}
		ai_acceptance = {
			condition = {
				base =10
			}
		}

		ai_desire = {
			base = 1

		}
	}
}
