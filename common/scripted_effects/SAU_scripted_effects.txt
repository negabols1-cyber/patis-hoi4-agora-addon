SAU_find_best_democratic_ally = {
	if = {
		limit = {
			FRA = {
				exists = yes
				has_government = democratic
				has_capitulated = no
				OR = {
					is_faction_leader = yes
					is_in_faction = no
				}
			}
		}
		set_variable = {
			SAU.SAU_best_dem_ally = FRA
		}
	}
	else_if = {
		limit = {
			ENG = {
				exists = yes
				is_faction_leader = yes
				has_government = democratic
			}
		}
		set_variable = {
			SAU.SAU_best_dem_ally = ENG
		}
	}
	else_if = {
		limit = {
			USA = {
				exists = yes
				is_faction_leader = yes
				has_government = democratic
			}
		}
		set_variable = {
			SAU.SAU_best_dem_ally = USA
		}
	}
	else = {
		random_other_country = {
			limit = {
				is_major = yes
				is_faction_leader = yes
				exists = yes
				has_government = democratic
				NOT = { has_war_with = ROOT }
			}
			SAU = {
				set_variable = {
					SAU_best_dem_ally = PREV
				}
			}
		}
	}
}
