		limit = {

				has_country_flag = LIB_national_education_1
			}
		}
		set_country_flag = LIB_national_education_1
	}
	else_if = {
		limit = {
			NOT = {
				has_country_flag = LIB_national_education_2
			}
		}

	}
	else_if = {
		limit = {
			NOT = {
				has_country_flag = LIB_national_education_3
			}
		}
		set_country_flag = LIB_national_education_3
	}
	else_if = {
		limit = {
			NOT = {
				has_country_flag = LIB_national_education_4
			}
		}
		set_country_flag = LIB_national_education_4
	}
	else_if = {
		limit = {
