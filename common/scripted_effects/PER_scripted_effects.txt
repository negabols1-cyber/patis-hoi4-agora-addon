PER_check_inefficient_army = {
	if = {
		limit = {
			has_dynamic_modifier = { modifier = PER_inefficient_army_dmod }
		}
		if = {
			limit = {
				NOT = { has_country_flag = PER_inefficient_army_name_change_flag }
				check_variable = {
					var = PER_buffed_inefficient_army
					value = 3
					compare = greater_than_or_equals
				}
			}
			hidden_effect = {
				set_country_flag = PER_inefficient_army_name_change_flag
				remove_dynamic_modifier = { modifier = PER_inefficient_army_dmod }
				add_dynamic_modifier = { modifier = PER_iran_army_dmod }
			}
		}
	}
	else = {
		army_experience = 10
	}
}
