BEL_election_stability_debuff = {
	if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 1 }
		}
		add_stability = -0.02
	}
	else_if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 2 }
		}
		add_stability = -0.03
	}
	else_if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 3 }
		}
		add_stability = -0.04
		hidden_effect = {
			country_event = { id = belgium.61 days = 20 random_days = 10 }
		}
	}
	else_if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 4 }
		}
		add_stability = -0.06
	}
	else_if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 5 }
		}
		add_stability = -0.08
		add_ideas = BEL_postponed_election
		custom_effect_tooltip = BEL_state_of_emergency_declare_tt
		hidden_effect = {
			set_country_flag = BEL_state_of_emergency_flag
			set_country_flag = BEL_failed_to_form_a_government_flag
		}
	}
	else_if = {
		limit = {
			check_variable = { BEL_postpone_election_count > 6 }
		}
		add_stability = -0.1
	}
}

		max = 0.15
	}

		max = 0.15
	}

		value = BEL_integration_lux

		min = 0

	}

	clear_variable = BEL_integration_lux
}

#Change name of States and VPs when appropriate. Expected to be used inside on_state_control_changed. FROM.FROM is state / ROOT is new controller / FROM is former controller
BEL_change_name_of_state_or_vp = {
	if = { # Dunkerque / Lille // (Duunkerke / Rysel)
		limit = {
			FROM.FROM = {
				state = 29 #Nord-Pas-de-Calais
			}
		}
		if = { #BELGIUM CONTROLLER
			limit = {
				ROOT = { original_tag = BEL }
			}
			set_province_name = { id = 575 name = VICTORY_POINTS_575_FLA } #Duunkerke
			set_province_name = { id = 11483 name = VICTORY_POINTS_11483_FLA } #Rysel
		}
		else_if = { #FLANDERS CONTROLLER
			limit = {
				ROOT = { original_tag = FLA }
			}
			set_province_name = { id = 575 name = VICTORY_POINTS_575_FLA } #Duunkerke
			set_province_name = { id = 11483 name = VICTORY_POINTS_11483_FLA } #Rysel
		}
		else_if = { #OTHER CONTROLLER
			limit = {
				FROM = { original_tag = BEL }
			}
			reset_province_name = 575
			reset_province_name = 11483
		}
	}
	else_if = { # Charleroi / Liége // (Tchålerwè / Lîdje)
		limit = {
			FROM.FROM = {
				state = 34 #Wallonie
			}
		}
		if = { #WALLONIA CONTROLLER
			limit = {
				ROOT = { original_tag = WLL }
			}
			set_province_name = { id = 9536 name = VICTORY_POINTS_9536_WLL } #Tchålerwè
			set_province_name = { id = 11519 name = VICTORY_POINTS_11519_WLL } #Lîdje
		}
		else_if = { #OTHER CONTROLLER
			limit = {
				FROM = { original_tag = WLL }
			}
			reset_province_name = 9536
			reset_province_name = 11519
		}
	}
}
