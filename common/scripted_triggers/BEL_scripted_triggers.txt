				original_tag = BEL
				has_government = democratic
			}
		}

		has_completed_focus = BEL_rapid_mobilization
	}
	else = {
		hidden_trigger = { always = no }  #NEED TO HAVE AN ALWAYS AS AN END STATEMENT IF URE USING AN IF ELSE IN AN OR STATEMENT OTHERWISE THE EVALUATION BREAKS!!
	}
}

BEL_is_historical_ai = {
	OR = {
		AND = {
			is_historical_focus_on = yes
			has_game_rule = {
				rule = BEL_ai_behavior
				option = DEFAULT
			}
		}
		has_country_flag = BEL_AI_RANDOM_HISTORICAL
		has_game_rule = {
			rule = BEL_ai_behavior
			option = HISTORICAL
		}
	}
}

BEL_state_or_vp_name_should_change = {
	OR = {
		AND = {
			ROOT = { original_tag = BEL }
			NOT = { FROM = { original_tag = BEL } } #Not civil wars
		}
		AND = {
			FROM = { original_tag = BEL }
			NOT = { ROOT = { original_tag = BEL } } #Not civil wars
		}
	}
	OR = {
		FROM.FROM = { state = 29 } # Dunkerque / Lille // (Duunkerke / Rysel)
		FROM.FROM = { state = 34 } # Charleroi / Liége // (Tchålerwè / Lîdje)
	}
}
