PER_is_historical_ai = {
	OR = {
		AND = {
			is_historical_focus_on = yes
			has_game_rule = {
				rule = PER_ai_behavior
				option = DEFAULT
			}
		}
		has_country_flag = PER_AI_RANDOM_HISTORICAL
		has_game_rule = {
			rule = PER_ai_behavior
			option = HISTORICAL
		}
	}
}
