	on_war = {
		effect = {
			if = {
				limit = {
					FRA = {
						has_war_with = GER
					}
				}
				set_global_flag = UFRA_fra_ger_war_rhine
			}
			if = {
				limit = {
					FRA = {
						has_war = yes
						NOT = { has_global_flag = UFRA_at_war }
					}
				}
				add_french_aces = yes
			}
		}
	}

	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					original_tag = FRA
					has_completed_focus = UFRA_modern_empire
					FROM.FROM = {
						has_resistance = yes
						is_controlled_by = FRA
					}
				}
				FROM.FROM = {
					add_compliance = 5
				}
			}
			if = {
				limit = {
					original_tag = FRA
					has_completed_focus = UFRA_eternal_discipline
					FROM.FROM = {
						has_resistance = yes
						is_controlled_by = FRA
					}
				}
				FROM.FROM = {
					add_resistance = -5
				}
			}
		}
	}
