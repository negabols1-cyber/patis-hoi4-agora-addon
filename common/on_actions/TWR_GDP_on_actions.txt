on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = allow_gdp_ranking
						option = ALLOWED
					}
				}
				every_country = {
					if = {
						limit = {
							count_triggers = {
								amount = 5
								GER = { is_ai = no }
								FRA = { is_ai = no }
								ITA = { is_ai = no }
								ENG = { is_ai = no }
								SOV = { is_ai = no }
								JAP = { is_ai = no }
								USA = { is_ai = no }
								CHI = { is_ai = no }
							}
							OR = {
								#Majors
								original_tag = GER
								original_tag = FRA
								original_tag = ITA
								original_tag = ENG
								original_tag = SOV
								original_tag = JAP
								original_tag = USA
								original_tag = CHI
								#Super Minor
								original_tag = TUR
								original_tag = SWE
								original_tag = SPR
								original_tag = ROM
								original_tag = BRA
								is_ai = no
							}
							NOT = { is_in_array = { global.GDP_list_array = THIS } }
						}
						add_to_array = { global.GDP_list_array = THIS }
						set_country_flag = EF_avarage_GDP_player
						set_global_flag = EF_optimised_GDP
						add_dynamic_modifier = {
							modifier = EF_country_gdp
						}
						#elleme
						calculate_GDP = yes
						GDP_growth_calculation = yes
						GDP_growth_calculation = yes
					}
					if = {
						limit = {
							NOT = { has_global_flag = EF_optimised_GDP }
							NOT = { is_in_array = { global.GDP_list_array = THIS } }
						}
						add_to_array = { global.GDP_list_array = THIS }
						set_country_flag = EF_avarage_GDP_player
						add_dynamic_modifier = {
							modifier = EF_country_gdp
						}
						calculate_GDP = yes
						GDP_growth_calculation = yes
						GDP_growth_calculation = yes
					}
					if = {
						limit = {
							1 = {
								is_owned_by = ROOT
							}
						}
						update_position = yes
					}
					set_variable = { ROOT.war_crime_rate = 0 }
				}
			}
		}
	}
	on_monthly = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = allow_gdp_ranking
						option = ALLOWED
					}
				}
				calculate_GDP = yes
				GDP_growth_calculation = yes
			}
			if = {
				limit = {
					1 = {
						is_owned_by = ROOT
					}
				}
				update_position = yes
			}
			THIS = {
				if = {
					limit = {
						has_country_flag = EF_avarage_GDP_player
						NOT = {
							has_dynamic_modifier = {
								modifier = EF_country_gdp
							}
						}
					}
					add_dynamic_modifier = {
						modifier = EF_country_gdp
					}
				}
			}
		}
	}
}
