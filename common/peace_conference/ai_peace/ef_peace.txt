	ai_null_peace = {
		peace_action_type = { take_states liberate puppet force_government }
		enable = {
			if = {
				limit = {
					has_global_flag = EF_multiplayer_tools_available_flag
					has_game_rule = {
						rule = allow_player_led_peace_conference
						option = ALLOWED
					}
				}
				OR = {
					any_country = {
						is_in_peace_conference = yes

					}
					any_enemy_country = {
						is_in_peace_conference = yes
						any_enemy_country = {
							is_in_peace_conference = yes
							is_ai = no
						}
					}
				}
			}
			else = {
				always = no
			}
		}
