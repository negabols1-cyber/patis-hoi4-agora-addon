# Protocol City Name
defined_text = {
	name = GetProtocolCityName
	text = {
		trigger = {
			OR = {
				ENG = {
					controls_state = 447
					NOT = { 447 = { is_core_of = ENG } }
					OR = {
						NOT = { country_exists = EGY }
						AND = {
							country_exists = EGY
							EGY = { is_subject_of = ENG }
						}
					}
				}
				EGY = {
					controls_state = 447
					OR = {
						has_government = neutrality
						has_government = democratic
					}
				}
			}
		}
		localization_key = SAU_alexandria_protocol_tt
	}
	text = {
		localization_key = SAU_riyadh_protocol_tt
	}
}
