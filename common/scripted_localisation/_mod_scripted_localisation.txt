# Allows you to create your own dynamic keys
# to be called in localization.
# defined_text -> this is it, we're defining the text
# text -> a discrete entry that can be picked to display in loc.
# trigger -> determines if a text entry will be picked or not.
# (The triggers need to be valid for the scope the key is called in
# (eg Root or From.From).)
# localization_key -> points to the localization key
# that'll be used if trigger passes

defined_text = {	# Rio de Janeiro's ruling party name & text for decision category
	name = GetRioMajorityPartyText

	text = {
		trigger = {
			has_country_flag = RIO_majority_democratic
		}
		localization_key = GetRioMajorityPartyTextDemocratic
	}
	text = {
		trigger = {
			has_country_flag = RIO_majority_fascism
		}
		localization_key = GetRioMajorityPartyTextFascism
	}
	text = {
		trigger = {
			has_country_flag = RIO_majority_neutrality
		}
		localization_key = GetRioMajorityPartyTextNeutrality
	}
	text = {
		trigger = {
			has_country_flag = RIO_majority_communism
		}
		localization_key = GetRioMajorityPartyTextCommunism
	}
	text = {
		trigger = {
			NOT = {
				has_country_flag = RIO_majority_democratic
				has_country_flag = RIO_majority_fascism
				has_country_flag = RIO_majority_neutrality
				has_country_flag = RIO_majority_communism
			}
		}
		localization_key = GetRioMajorityPartyTextNone
	}
}
